chunk_score = []
for chunk in doc_chunks:
    score = sum(1 for word in chunk if word in question)
    chunk_score.append((score, chunk))
chunk_score.sort(reverse=True)
for score, chunk in chunk_score[:3]:
    if score >0:
        result["text_chunks"].append(chunk)
