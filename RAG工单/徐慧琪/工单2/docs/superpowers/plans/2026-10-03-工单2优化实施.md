```python
# -*- coding: utf-8 -*-
"""
工单编号：人工智能NLP-RAG-基于PDF文档的问答系统优化
模块：PDF 解析（MinerU pipeline 主解析 + 质量检测 + PyMuPDF 回退）

本机实测：
  - pipeline 后端可用；必须带 -f false（MFD 公式检测模型本机缺失且禁下载）。
  - vlm-auto-engine 后端与 transformers 5.15 不兼容（Qwen2VLConfig 无
    max_position_embeddings 属性），故不采用。
  - 解析失败自动回退 PyMuPDF，保证建库链路不被单点阻塞。
"""
from src import bootstrap  # noqa: F401  —— 必须最先导入

from __future__ import annotations

import json
import os
import subprocess
import sys
import time

from src import config

_MINERU_EXE = os.path.join(os.path.dirname(sys.executable), "Scripts", "mineru.exe")


def _mineru_env() -> dict:
    env = dict(os.environ)
    env["MINERU_MODEL_SOURCE"] = "local"     # 强制本地模型，禁止联网
    env["HF_HUB_OFFLINE"] = "1"
    env["TRANSFORMERS_OFFLINE"] = "1"
    env["PYTHONIOENCODING"] = "utf-8"
    env["TOKENIZERS_PARALLELISM"] = "false"
    return env


def run_mineru(pdf_path: str, out_dir: str, start: int | None = None,
               end: int | None = None, timeout: int | None = None) -> dict:
    """调用 MinerU CLI（pipeline 后端，关闭公式解析）。不抛异常，返回结构化结果。"""
    timeout = timeout or config.MINERU_TIMEOUT_S
    exe = _MINERU_EXE if os.path.isfile(_MINERU_EXE) else "mineru"
    cmd = [exe, "-p", pdf_path, "-o", out_dir, "-b", config.MINERU_BACKEND,
           "-f", "true" if config.MINERU_FORMULA else "false"]
    if start is not None:
        cmd += ["-s", str(start)]
    if end is not None:
        cmd += ["-e", str(end)]
    t0 = time.time()
    try:
        proc = subprocess.run(cmd, env=_mineru_env(), capture_output=True, text=True,
                              encoding="utf-8", errors="replace", timeout=timeout)
    except subprocess.TimeoutExpired:
        return {"ok": False, "md_path": None, "content_list_path": None,
                "elapsed_s": time.time() - t0, "error": f"timeout>{timeout}s"}
    except Exception as exc:  # noqa: BLE001
        return {"ok": False, "md_path": None, "content_list_path": None,
                "elapsed_s": time.time() - t0, "error": repr(exc)}

    stem = os.path.splitext(os.path.basename(pdf_path))[0]
    md_path = cl_path = None
    for root, _dirs, files in os.walk(out_dir):
        for name in files:
            if name == f"{stem}.md":
                md_path = os.path.join(root, name)
            elif name == f"{stem}_content_list.json":
                cl_path = os.path.join(root, name)
    ok = proc.returncode == 0 and cl_path is not None
    return {"ok": ok, "md_path": md_path, "content_list_path": cl_path,
            "elapsed_s": time.time() - t0,
            "error": None if ok else (proc.stderr or proc.stdout or "")[-3000:]}


def _join_caption(item: dict) -> str:
    caps = list(item.get("image_caption") or []) + list(item.get("table_caption") or [])
    return " ".join(str(c) for c in caps if c).strip()


def normalize_item(item: dict) -> dict | None:
    """MinerU 原始条目 → 内部结构；不可解析返回 None。"""
    t = item.get("type")
    page = int(item.get("page_idx") or 0)
    if t == "text":
        text = (item.get("text") or "").strip()
        return {"type": "text", "text": text, "page_idx": page,
                "text_level": item.get("text_level"), "caption": []} if text else None
    if t == "table":
        body = (item.get("table_body") or "").strip()
        cap = _join_caption(item)
        text = (cap + "\n" + body).strip() if cap else body
        if not text:
            return None
        return {"type": "table", "text": text, "page_idx": page, "text_level": None,
                "caption": [cap] if cap else [], "img_path": item.get("img_path")}
    if t == "equation":
        text = (item.get("text") or "").strip()
        return {"type": "equation", "text": text, "page_idx": page,
                "text_level": None, "caption": []} if text else None
    if t == "image":
        cap = _join_caption(item)
        return {"type": "image", "text": cap or "[图片]", "page_idx": page,
                "text_level": None, "caption": [cap] if cap else [],
                "img_path": item.get("img_path")}
    return None


def load_content_list(path: str) -> list[dict]:
    with open(path, encoding="utf-8") as fh:
        raw = json.load(fh)
    return [n for n in (normalize_item(it) for it in raw) if n is not None]


def quality_report(items: list[dict], page_count: int) -> dict:
    """低文本页（< OCR_MIN_CHARS_PER_PAGE 字符）提示需要 OCR 兜底。"""
    per_page: dict[int, int] = {}
    for it in items:
        if it["type"] == "text":
            per_page[it["page_idx"]] = per_page.get(it["page_idx"], 0) + len(it["text"])
    return {
        "pages_with_text": sum(1 for p in range(page_count) if per_page.get(p, 0) > 0),
        "low_text_pages": [p for p in range(page_count)
                           if per_page.get(p, 0) < config.OCR_MIN_CHARS_PER_PAGE],
        "n_tables": sum(1 for i in items if i["type"] == "table"),
        "n_equations": sum(1 for i in items if i["type"] == "equation"),
        "n_images": sum(1 for i in items if i["type"] == "image"),
    }


def _page_count(pdf_path: str) -> int:
    import fitz
    doc = fitz.open(pdf_path)
    n = doc.page_count
    doc.close()
    return n


def _parse_with_pymupdf(pdf_path: str) -> list[dict]:
    """回退解析：PyMuPDF 文本 + 表格（find_tables）。"""
    import fitz

    doc = fitz.open(pdf_path)
    items: list[dict] = []
    for pno in range(doc.page_count):
        page = doc[pno]
        text = page.get_text("text") or ""
        for para in [p.strip() for p in text.split("\n\n") if p.strip()]:
            items.append({"type": "text", "text": para, "page_idx": pno,
                          "text_level": None, "caption": []})
        try:
            tables = page.find_tables()
        except Exception:  # noqa: BLE001
            tables = []
        for tbl in tables or []:
            try:
                md = (tbl.to_markdown() or "").strip()
            except Exception:  # noqa: BLE001
                continue
            if md:
                items.append({"type": "table", "text": md, "page_idx": pno,
                              "text_level": None, "caption": []})
    doc.close()
    return items


def parse_pdf(pdf_path: str, out_dir: str, reuse: bool = True,
              force_fallback: bool = False) -> dict:
    """解析入口：缓存复用 → MinerU → PyMuPDF 回退。"""
    stem = os.path.splitext(os.path.basename(pdf_path))[0]
    cached_cl = os.path.join(out_dir, stem, "auto", f"{stem}_content_list.json")
    n = _page_count(pdf_path)

    if reuse and not force_fallback and os.path.isfile(cached_cl):
        return {"source": os.path.basename(pdf_path), "parser": "mineru",
                "items": load_content_list(cached_cl),
                "md_path": os.path.join(os.path.dirname(cached_cl), f"{stem}.md"),
                "page_count": n}

    if not force_fallback:
        res = run_mineru(pdf_path, out_dir)
        if res["ok"]:
            return {"source": os.path.basename(pdf_path), "parser": "mineru",
                    "items": load_content_list(res["content_list_path"]),
                    "md_path": res["md_path"], "page_count": n}
        print(f"[pdf_parser] MinerU 失败：{(res['error'] or '')[-300:]}", flush=True)

    print("[pdf_parser] 回退 PyMuPDF 解析", flush=True)
    return {"source": os.path.basename(pdf_path), "parser": "pymupdf",
            "items": _parse_with_pymupdf(pdf_path), "md_path": None,
            "page_count": n}
```

## Task 4: 分块模块（标题层级 + 表格独立 + 父子块）★核心

**Files:** Create `src/chunker.py`、`tests/test_chunker.py`

**Interfaces:**
- Consumes: `src.config`（`CHILD_*`、`PARENT_MAX_CHARS`、`TABLE_MAX_CHARS`、`SECTION_MIN_CHARS`）、`src.pdf_parser` 条目
- Produces:
  - `build_chunks(items: list[dict], source: str) -> list[dict]`，chunk 结构：
    ```python
    {
      "chunk_id": "c000123",          # 子块唯一 id（稳定、可复现）
      "text": str,                    # 子块（检索单位，向量化）
      "parent_id": "s000042",         # 父块 id
      "parent_text": str,             # 父块（LLM 上下文单位）
      "parent_page_idx": int,
      "page_idx": int,                # 子块起始页
      "heading_path": list[str],      # 标题层级路径
      "source": str,
      "block_type": "text|table|equation|image",
      "seq": int,                     # 文档内顺序（相邻块扩展用）
    }
    ```
  - `iter_sections(items) -> list[dict]`（内部：按 text_level 切节，`{"heading_path", "items", "page_idx"}`）

**算法（完整实现）：**
1. 遍历 items，维护 `heading_stack`（list[(level, title)]）。`type=="text" and text_level` 视为标题 → 出栈至 level-1 再入栈，同时**结束当前 section**、开启新 section（heading_path = 栈中标题）。
2. 表格/公式/图片条目 → 独立成块：不进 section 正文流，`parent_id` 归属当前 section，`block_type` 标注；表格超 `TABLE_MAX_CHARS` 时截断（保留首行表头 + 尾行 + "…"标记）。
3. section 内正文条目按序累积为子块：贪心累积到 ≥`CHILD_MAX_CHARS`（且 ≥`CHILD_MIN_CHARS`）时切块；块内句子边界优先（`。！？；\n`），不切断句子。
4. 父块 = section 全文（标题 + 正文 + 表格文本），超 `PARENT_MAX_CHARS` 截断。
5. 短 section（< `SECTION_MIN_CHARS`）与下一节合并为同一父块（避免碎片化）。
6. `chunk_id`/`parent_id` 由全局递增序号生成（`c%06d`/`s%06d`），保证可复现。
