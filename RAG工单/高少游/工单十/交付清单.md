docker inspect --format '{{.State.Health.Status}}' finqa-app   # healthy
docker exec finqa-app python deploy/smoke_test.py          # 端到端问答
bash deploy/verify.sh                                      # 容器管理验收（Linux）
powershell -File deploy/verify.ps1                         # 容器管理验收（Windows）
