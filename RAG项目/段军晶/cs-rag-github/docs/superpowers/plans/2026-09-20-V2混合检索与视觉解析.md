```python
# -*- coding: utf-8 -*-
"""
RRF（Reciprocal Rank Fusion）融合模块

职责：
    把多个检索通道的**有序结果列表**融合为单一排序。

设计要点：
    1. 纯函数、无 I/O、无副作用 —— 便于单元测试与复用
    2. 融合基于**名次**而非分数，因此天然免疫各通道分数尺度不一致的问题
       （稠密通道为余弦相似度，稀疏通道为内积，二者不可直接比较）
    3. k 值取自配置 settings.rrf_k（默认 60），不硬编码

融合公式：
    score(chunk) = Σ_c  1 / (k + rank_c(chunk))

    其中 rank_c 为该 chunk 在第 c 通道中的名次（从 1 开始）；
    未出现在某通道则不计该项。
"""

from __future__ import annotations

from typing import Dict, List, Optional, Sequence, Tuple

from backend.config import settings

FusedItem = Tuple[str, float]


def fuse(
    channels: Sequence[Sequence[str]],
    *,
    k: Optional[int] = None,
    top_k: int = 5,
) -> List[FusedItem]:
    """
    按 RRF 融合多个通道的检索结果。

    参数：
        channels : 各通道的 chunk_id 有序列表（按名次从优到劣）。
                   允许空列表，允许各通道长度不同。
        k        : RRF 平滑常数。None 时取 settings.rrf_k（默认 60）。
        top_k    : 返回结果条数。

    返回：
        [(chunk_id, rrf_score), ...] 按分数降序；
        分数相同时按 chunk_id 升序，保证结果可复现。
    """
    if k is None:
        k = settings.rrf_k
    if k <= 0:
        raise ValueError(f"RRF 的 k 必须为正整数，当前为 {k}")
    if top_k <= 0:
        return []

    scores: Dict[str, float] = {}
    for channel in channels:
        seen = set()
        for rank, chunk_id in enumerate(channel, start=1):
            # 同一通道内重复出现的块只记首次名次，避免重复计分
            if not chunk_id or chunk_id in seen:
                continue
            seen.add(chunk_id)
            scores[chunk_id] = scores.get(chunk_id, 0.0) + 1.0 / (k + rank)

    ranked = sorted(scores.items(), key=lambda kv: (-kv[1], kv[0]))
    return ranked[:top_k]
```

```python
# ---------------------------------------------------------------------------
# 稀疏编码（V2 混合检索）
# ---------------------------------------------------------------------------

# BGE-M3 基于 XLM-Roberta，其特殊 token 为：<s>=0、<pad>=1、</s>=2。
# 实测这些 token 也会获得 0.16~0.20 的权重，但它们在**任何**文本中都出现，
# 只贡献常量偏置、无区分度，属纯噪声，必须显式排除。
_SPECIAL_TOKEN_IDS = frozenset({0, 1, 2})


def _to_sparse_dict(
    weights: Any,
    input_ids: Any,
    attention_mask: Any,
) -> Dict[int, float]:
    """
    把 token 级权重压缩为 {token_id: weight} 稀疏字典。

    规则：
        1. padding 位置（attention_mask=0）跳过
        2. 权重 <= 0 的 token 跳过（ReLU 后仍可能为 0）
        3. 特殊 token 跳过（见 _SPECIAL_TOKEN_IDS）
        4. 同一 token_id 多次出现时取最大权重

    这是纯函数，不依赖模型，便于单元测试。
    """
    result: Dict[int, float] = {}
    for tok_id, weight, mask in zip(
        input_ids.tolist(), weights.tolist(), attention_mask.tolist()
    ):
        if mask == 0 or weight <= 0:
            continue
        if tok_id in _SPECIAL_TOKEN_IDS:
            continue
        if weight > result.get(tok_id, 0.0):
            result[tok_id] = float(weight)
    return result
```

```python
    # ----------------------------------------------------------------
    # 稀疏权重头
    # ----------------------------------------------------------------

    def _load_sparse_linear(self) -> Any:
        """
        加载 BGE-M3 的稀疏权重头 sparse_linear.pt。

        该文件仅 3.5KB，是 BGE-M3 官方的 lexical weights 线性层。
        直接加载可避免引入 FlagEmbedding 依赖（详见 ADR-014）。
        """
        if self._sparse_linear is not None:
            return self._sparse_linear

        with self._lock:
            if self._sparse_linear is not None:
                return self._sparse_linear

            from pathlib import Path

            import torch

            path = Path(self.model_path) / "sparse_linear.pt"
            if not path.exists():
                raise EmbedderNotAvailableError(
                    f"BGE-M3 稀疏权重文件不存在：{path}\n"
                    f"V2 混合检索需要该文件（sparse_linear.pt）"
                )

            state = torch.load(str(path), map_location="cpu")
            linear = torch.nn.Linear(state["weight"].shape[1], state["weight"].shape[0])
            linear.load_state_dict(state)
            linear.eval()
            linear.to(self.device)

            self._sparse_linear = linear
            logger.info("BGE-M3 稀疏权重头加载完成：%s", path)
            return linear
```

- [ ] **Step 4: 实现一次 forward 双输出与 encode_sparse**

替换 `backend/embedder.py` 中现有的 `encode_sparse`（当前为 `raise NotImplementedError` 的桩）：

```python
    def encode_dense_and_sparse(
        self,
        texts: Sequence[str],
        *,
        batch_size: Optional[int] = None,
        show_progress: bool = False,
    ) -> Tuple[List[List[float]], List[Dict[int, float]]]:
        """
        一次 forward 同时产出稠密向量与稀疏权重。

        为什么要合并为一次 forward：
            BGE-M3 的稠密与稀疏共用同一个编码器，只有输出头不同。
            分两次调用会让 CPU 环境下的推理耗时翻倍。
            已验证手动 CLS 池化的结果与 sentence-transformers 完全一致。

        返回：(稠密向量列表, 稀疏字典列表)，两者与输入等长且一一对应。
        """
        if not texts:
            return [], []

        import torch

        model = self._load()
        auto_model = model[0].auto_model          # 底层 XLMRobertaModel
        tokenizer = model.tokenizer
        sparse_linear = self._load_sparse_linear()

        bs = batch_size or self.batch_size
        auto_model.eval()
        sparse_linear.eval()

        dense_all: List[List[float]] = []
        sparse_all: List[Dict[int, float]] = []

        for start in range(0, len(texts), bs):
            batch = list(texts[start : start + bs])
            encoded = tokenizer(
                batch,
                padding=True,
                truncation=True,
                max_length=self.max_length,
                return_tensors="pt",
            )
            encoded = {k: v.to(self.device) for k, v in encoded.items()}

            with torch.no_grad():
                hidden = auto_model(**encoded).last_hidden_state   # (B, L, 1024)
                # 稠密：CLS 池化 + L2 归一化（与 1_Pooling/config.json 一致）
                cls = hidden[:, 0]
                dense = torch.nn.functional.normalize(cls, p=2, dim=1)
                # 稀疏：ReLU 后的 token 级权重，屏蔽 padding
                raw = sparse_linear(hidden).squeeze(-1)            # (B, L)
                weights = torch.relu(raw * encoded["attention_mask"])

            dense_all.extend(dense.cpu().numpy().tolist())
            for i in range(len(batch)):
                sparse_all.append(
                    _to_sparse_dict(
                        weights[i], encoded["input_ids"][i], encoded["attention_mask"][i]
                    )
                )

        return dense_all, sparse_all

    def encode_sparse(
        self,
        texts: Sequence[str],
        *,
        batch_size: Optional[int] = None,
    ) -> List[Dict[int, float]]:
        """

        调用方若同时需要稠密向量，应直接调用 encode_dense_and_sparse 以避免重复推理。
        """
        _, sparse = self.encode_dense_and_sparse(texts, batch_size=batch_size)
        return sparse
```

```python
def search_sparse(
    query_sparse: Dict[int, float],
    *,
    top_k: Optional[int] = None,
    output_fields: Optional[List[str]] = None,
    name: Optional[str] = None,
) -> List[Dict[str, Any]]:
    """
    稀疏向量检索（V2 混合检索的稀疏通道）。

    度量方式为 IP（内积）—— BGE-M3 的 lexical weights 非负且不归一化，
    内积即为加权词匹配得分，这是 BGE-M3 稀疏表示的标准用法。

    返回结构与 search_dense() 对称，含 chunk_id / score / doc_id / page_no / content。
    注意：这里的 page_no / content 来自 Milvus 的冗余标量字段，
    仅用于 Trace 日志回显；对外溯源信息一律以 MySQL 为准（ADR-005）。
    """
    if not query_sparse:
        return []

    name = name or settings.milvus_collection
    client = get_client()
    limit = top_k or settings.retrieve_top_k

    results = client.search(
        collection_name=name,
        data=[query_sparse],
        anns_field="sparse_embedding",
        search_params={"metric_type": "IP", "params": {}},
        limit=limit,
        output_fields=output_fields or ["chunk_id", "doc_id", "page_no", "content"],
    )

    hits: List[Dict[str, Any]] = []
    for hit in (results[0] if results else []):
        entity = hit.get("entity", {}) or {}

        # ★ 与 search_dense 保持同样的主键取值兼容 ★
        # pymilvus 不同版本下主键可能出现在 hit 顶层（字段名同 schema）或 entity 内，
        # 依次尝试多个来源，避免取不到主键导致下游元数据回填全部失败。
        chunk_id = (
            hit.get("chunk_id")
            or hit.get("id")
            or entity.get("chunk_id")
        )

        hits.append({
            "chunk_id": chunk_id,
            "score": float(hit.get("distance", 0.0)),
            "doc_id": entity.get("doc_id"),
            "page_no": entity.get("page_no"),
            "content": entity.get("content", ""),
        })
    return hits
```

**缺陷 A：Milvus 侧的 content 截断对中文失效**

原实现 `str(contents[i])[:_MAX_LEN_CONTENT]` 按**字符**切片，而 Milvus 的
`VARCHAR(max_length=8192)` 按 **UTF-8 字节**计数。中文 8192 字符 = 24576 字节，
Milvus 直接拒绝整条插入（`MilvusException(code=1100)`），触发阈值是 **2730 个中文字符**。

新增字节截断辅助函数（放在 `_client` 定义之后）：

```python
def _truncate_utf8(text: str, max_bytes: int, *, field: str = "") -> str:
    """
    按 UTF-8 **字节数**截断字符串。

    为什么不能用 text[:max_bytes]：
        Milvus 的 VARCHAR(max_length=N) 以 **UTF-8 字节**计数，而 Python 的
        字符串切片按**字符**计数。一个中文字符占 3 字节，因此对中文内容做
        text[:8192] 实际会产生 24576 字节 —— 超出字段上限，导致
        MilvusException(code=1100) 并使**整条插入失败**（不是静默截断）。

    errors="ignore" 用于避免在多字节字符中间切断时抛 UnicodeDecodeError。
    """
    encoded = text.encode("utf-8")
    if len(encoded) <= max_bytes:
        return text
    truncated = encoded[:max_bytes].decode("utf-8", errors="ignore")
    logger.warning(
        "字段超长已按 UTF-8 字节截断 | 字段=%s | 原长=%d 字节 | 截断至=%d 字节",
        field or "(未知)", len(encoded), max_bytes,
    )
    return truncated
```

**缺陷 B：写入后不 flush，新数据静默不可检索**

实测：同一进程内 `insert` 后立即 `search`，**不 flush 命中数为 0**，flush 后立刻命中。
原代码库**没有任何 flush 调用**。V1 未暴露此问题仅因为知识库提前一天灌入、已被自动 flush；
但 V2 的 Task 8「重建后立刻评测」必然踩到，且**不报错、只是少召回**。

新增辅助函数：

```python
def flush_collection(name: Optional[str] = None) -> None:
    """
    强制落盘，使已写入的数据**立即可被检索到**。

    为什么需要显式 flush：
        Milvus 在默认一致性级别下，新插入的数据要等自动 flush（周期性触发）
        之后才进入可检索状态。实测：同一进程内先 insert 再 search，
        不 flush 时命中数为 **0**；flush 后立刻命中。
        这个失败是**静默的** —— 不报错，只是少召回甚至完全查不到，
        属于最难察觉的一类故障。因此写入完成后必须显式 flush。
    """
    name = name or settings.milvus_collection
    try:
        get_client().flush(collection_name=name)
    except Exception as exc:
        # flush 失败不应让整批写入失败：数据已写入，只是可见性延迟
        logger.warning("Milvus flush 失败，数据可能延迟可见 | %s | %s", name, exc)
```

并在 `insert_vectors()` 的写入循环之后、返回之前调用它：

```python
    # 写入后立即 flush：否则新数据在 Milvus 自动 flush 前不可检索，
    # 而该失败是静默的（不报错、只少召回），会直接影响重建后立刻进行的评测。
    if written:
        flush_collection(name)

    logger.info("Milvus 写入向量完成：%d 条 -> %s", written, name)
    return written
```

```python
# 视觉描述在图注之后的拼接标记（前导换行符计入长度）
_VISION_MARK = "\n[视觉解析] "


def compose_block_content(caption: str, vision_desc: str, *, max_chars: int) -> str:
    """
    拼接独立块（表格 / 图片）的 content，并做保护性截断。

    为什么需要保护性截断：
        直接对「图注 + 描述」做硬切片，可能把描述拦腰截断，
        甚至在图注很长时把描述**整段切掉**，导致视觉解析白做。
        本函数保证：图注优先完整保留，剩余预算分配给描述。

    参数：
        caption     : 原始图注（或表格文本）
        vision_desc : 视觉模型解析出的描述，可为空
        max_chars   : content 字符上限（来自 settings.chunk_max_chars）

    返回：长度不超过 max_chars 的 content 字符串。
    """
    caption = (caption or "").strip()
    vision_desc = (vision_desc or "").strip()

    if not vision_desc:
        return caption[:max_chars]
    if not caption:
        return vision_desc[:max_chars]

    budget = max_chars - len(caption) - len(_VISION_MARK)
    if budget <= 0:
        # 图注已占满预算，放弃描述，保证图注完整
        return caption[:max_chars]

    if len(vision_desc) <= budget:
        return caption + _VISION_MARK + vision_desc

    truncated = vision_desc[: max(0, budget - 1)]
    return caption + _VISION_MARK + truncated + "…"
```

```python
    MinerU 只提取出图注（如「图 2 软件质量模型」），图内的结构信息
    在 V1 中完全缺失 —— 这是基线报告 4.5 记录的核心弱点。

用法：
    python -m scripts.vision_caption              # 增量解析（已解析的图跳过）
    python -m scripts.vision_caption --force      # 强制重新解析
    python -m scripts.vision_caption --dry-run    # 只列出待解析图片，不调用 API

产物：
    data/parsed/<stem>.vision.json
    以图片文件名为键，便于分块阶段查找；幂等，重跑不会重复调用 API。
"""

from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path
from typing import Any, Dict, List, Optional

from backend.config import settings
from backend.logging_config import get_logger, setup_logging

logger = get_logger(__name__)


def _image_root(parsed_dir: Path, stem: str) -> Optional[Path]:
    """
    定位某份文档的图片根目录。

    MinerU 的目录结构为：
        data/parsed/<stem>/<原始文件名>/ocr/images/*.jpg
    而 .parsed.json 中的 img_path 是相对于 <原始文件名>/ocr 的，
    形如 "images/xxx.jpg"。
    """
    doc_dir = parsed_dir / stem
    if not doc_dir.is_dir():
        return None
    subdirs = [d for d in doc_dir.iterdir() if d.is_dir()]
    if not subdirs:
        return None
    return subdirs[0] / "ocr"


def _collect_images(parsed_dir: Path, stem: str) -> List[Dict[str, Any]]:
    """从 .parsed.json 中收集图片块，并把 img_path 解析为绝对路径"""
    parsed_file = parsed_dir / f"{stem}.parsed.json"
    if not parsed_file.exists():
        return []

    data = json.loads(parsed_file.read_text(encoding="utf-8"))
    root = _image_root(parsed_dir, stem)

    images: List[Dict[str, Any]] = []
    for block in data.get("blocks", []):
        if block.get("type") != "image":
            continue
        img_path = block.get("img_path")
        if not img_path or root is None:
            continue
        full = root / img_path
        if not full.exists():
            logger.warning("图片文件不存在，跳过 | %s", full)
            continue
        images.append(
            {
                "image_path": full,
                "image_name": Path(img_path).name,
                "caption": (block.get("caption") or block.get("text") or "").strip(),
                "page_no": int(block.get("page_no", 0) or 0),
                "section_title": (block.get("section_title") or "").strip(),
            }
        )
    return images


def _describe_all(
    images: List[Dict[str, Any]], cache: Dict[str, Any], *, force: bool
) -> Dict[str, Any]:
    """逐张调用视觉模型，单张失败不中断整体流程"""
    from backend.llm_client import get_vision_llm

    vision = get_vision_llm()
    if not vision.available:
        logger.error(
            "视觉模型不可用：请检查 .env 中的 VISION_ENABLED 与 VISION_API_KEY"
        )
        return cache

    for item in images:
        name = item["image_name"]
        if not force and cache.get(name, {}).get("status") == "ok":
            logger.info("已有描述，跳过 | %s", name)
            continue

        logger.info(
            "解析图片 | %s | 第 %s 页 | %s",
            name, item["page_no"], item["caption"][:30],
        )
        try:
            desc = vision.describe_image(
                item["image_path"],
                caption=item["caption"],
                context=item["section_title"],
            )
            status = "ok" if (desc or "").strip() else "failed"
            cache[name] = {
                "caption": item["caption"],
                "desc": (desc or "").strip(),
                "page_no": item["page_no"],
                "section_title": item["section_title"],
                "status": status,
            }
            logger.info("解析完成 | %s | 描述长度=%d | 状态=%s", name, len(cache[name]["desc"]), status)
        except Exception as exc:
            # 单张失败不得中断整库构建，退回「仅图注」即 V1 行为
            logger.warning("解析失败，该图退回仅图注 | %s | %s", name, exc)
            cache[name] = {
                "caption": item["caption"],
                "desc": "",
                "page_no": item["page_no"],
                "section_title": item["section_title"],
                "status": "failed",
            }
    return cache


def process_one(parsed_dir: Path, stem: str, *, force: bool = False, dry_run: bool = False) -> int:
    """处理单份文档，返回成功解析的图片数"""
    images = _collect_images(parsed_dir, stem)
    if not images:
        return 0

    out_file = parsed_dir / f"{stem}.vision.json"

    if dry_run:
        print(f"\n【{stem}】待解析图片 {len(images)} 张")
        for item in images:
            print(f"  第 {item['page_no']:>3} 页 | {item['image_name'][:16]}… | {item['caption'][:40]}")
        return 0

    cache: Dict[str, Any] = {}
    if out_file.exists() and not force:
        cache = json.loads(out_file.read_text(encoding="utf-8"))

    cache = _describe_all(images, cache, force=force)
    out_file.write_text(
        json.dumps(cache, ensure_ascii=False, indent=2), encoding="utf-8"
    )
    ok = sum(1 for v in cache.values() if v.get("status") == "ok")
    logger.info("视觉解析完成 | %s | 成功 %d / 共 %d", stem, ok, len(images))
    return ok


def main(argv: Optional[List[str]] = None) -> int:
    setup_logging()
    parser = argparse.ArgumentParser(description="离线视觉解析：把图表语义补进知识库")
    parser.add_argument("--force", action="store_true", help="强制重新解析已有描述的图片")
    parser.add_argument("--dry-run", action="store_true", help="只列出待解析图片，不调用 API")
    args = parser.parse_args(argv)

    parsed_dir = Path(settings.parsed_path)
    parse_files = sorted(parsed_dir.glob("*.parsed.json"))
    if not parse_files:
        logger.error("未找到解析产物，请先执行 scripts.pdf_parse | %s", parsed_dir)
        return 1

    total = 0
    for f in parse_files:
        stem = f.name[: -len(".parsed.json")]
        total += process_one(parsed_dir, stem, force=args.force, dry_run=args.dry_run)

    if not args.dry_run:
        logger.info("全部完成 | 成功解析图片 %d 张", total)
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
def _standalone_block(
    block: Dict[str, Any],
    vision_cache: Optional[Dict[str, Dict[str, Any]]] = None,
) -> Dict[str, Any]:
    """
    把表格块 / 图片块转换为独立 chunk 记录（不参与切分）。

    V2 变化：
        图片块若存在视觉解析描述，会以「图注 + [视觉解析] 描述」的形式
        并入 content，使图内语义可被检索与生成使用。
        拼接与截断由 compose_block_content 保证不超上限、图注完整。
    """
    block_type = block.get("type", "text")
    text = (block.get("text") or "").strip()
    page_no = int(block.get("page_no", 1))

    if block_type == "image":
        # 用图片文件名去视觉缓存中查描述
        image_name = Path(block.get("img_path") or "").name
        entry = (vision_cache or {}).get(image_name) or {}
        vision_desc = entry.get("desc", "") if entry.get("status") == "ok" else ""
        if not text:
            text = f"[图片] 位于第 {page_no} 页的插图"
        content = compose_block_content(
            text, vision_desc, max_chars=settings.chunk_max_chars
        )
        if len(text) + len(_VISION_MARK) + len(vision_desc) > settings.chunk_max_chars:
            logger.warning(
                "图片块描述超长已截断 | 第 %s 页 | 原长=%d | 上限=%d",
                page_no,
                len(text) + len(_VISION_MARK) + len(vision_desc),
                settings.chunk_max_chars,
            )
    else:
        content = (text or f"[表格] 位于第 {page_no} 页的表格")[
            : settings.chunk_max_chars
        ]

    return {
        "content": content,
        "page_no": page_no,
        # 表格/图片独立成块，页码归属唯一，不会跨页
        "page_nums": [page_no],
        "section_title": block.get("section_title", ""),
        "content_type": block_type,
    }
```

```python
# -*- coding: utf-8 -*-
"""
V2 检索链路：稠密 + 稀疏混合检索 + RRF 融合

与 V1 的差异仅在**检索环节**：
    V1：稠密 top-5
    V2：稠密 top-20 + 稀疏 top-20 → RRF 融合 → top-5

会话管理、查询缓存、提示词、生成、溯源、记录等环节与 V1 保持一致，
其中公共部分直接复用 v1_pipeline 中已存在的实现（不修改该文件）。

设计依据见 docs/superpowers/specs/2026-09-20-V2混合检索与视觉解析-design.md
"""

from __future__ import annotations

from typing import Any, Dict, List, Optional

from backend.config import settings
from backend.db import milvus_client, mysql, redis_client
from backend.embedder import get_embedder
from backend.llm_client import get_llm

from backend.rag_pipeline.v1_pipeline import (
    SYSTEM_PROMPT,
    USER_TEMPLATE,
    CONTEXT_TEMPLATE,
    V1Pipeline,
)

logger = get_logger(__name__)
```

**本任务的核心是 `retrieve()` 方法**，其完整实现见下面类定义中的方法体。实现思路（六步）：

1. **一次 forward** 同时得到稠密向量与稀疏权重
2. **两路独立检索**：`search_dense` 与 `search_sparse` 各取 `candidate_k`（20）条
3. **两路分别落 Trace**（必须在融合前，否则无法归因）
4. **RRF 融合**：`rrf.fuse([dense_ids, sparse_ids], top_k=final_k)`
5. **融合结果与两路名次落 Trace**
6. **回填 MySQL 元数据**，组装成与 V1 相同结构的结果列表

**类与单例（文件末尾）**：

```python
class V2Pipeline(V1Pipeline):
    """
    V2 链路：继承 V1Pipeline 复用会话/缓存/生成/溯源等全部公共环节，
    仅覆盖 retrieve() 为混合检索。

    为什么继承而不是复制：
        V1Pipeline.answer() 在内部调用 self.retrieve(question, top_k=top_k)
        （v1_pipeline.py:139），因此覆盖 retrieve() 即可让整条 answer 链路
        自动走混合检索 —— 无需重写 answer()，也无需复制任何公共逻辑。

    版本标识：
        V1Pipeline 用类属性 PIPELINE_NAME 生成响应中的 pipeline 字段
        （v1_pipeline.py:337 的 self.PIPELINE_NAME），此处覆盖为 "v2"。
    """

    PIPELINE_NAME = "v2"

    def retrieve(
        self,
        question: str,
        *,
        top_k: Optional[int] = None,
    ) -> List[Dict[str, Any]]:
        """
        V2 混合检索：稠密 + 稀疏 → RRF 融合。

        返回结构与 V1 相同（含 chunk_id / score / page_no / file_name /
        page_nums / content / content_type / section_title），
        其中 score 为 RRF 融合分数。
        """
        final_k = top_k or settings.retrieve_top_k
        candidate_k = settings.retrieve_candidate_k

        # ① 一次 forward 同时得到稠密向量与稀疏权重
        embedder = get_embedder()
        dense_vectors, sparse_vectors = embedder.encode_dense_and_sparse([question])
        query_dense = dense_vectors[0] if dense_vectors else []
        query_sparse = sparse_vectors[0] if sparse_vectors else {}

        # ② 两路独立检索（本地连接，两次往返开销在毫秒级）
        dense_hits = milvus_client.search_dense(query_dense, top_k=candidate_k)
        sparse_hits = milvus_client.search_sparse(query_sparse, top_k=candidate_k)

        # ③ 两路结果分别落 Trace —— 必须在融合前记录，
        #    否则无法回答「某块是靠稠密还是靠稀疏被召回的」
        #    log_retrieval_trace 直接接收 hits 列表（元素含 chunk_id/score/page_no），
        #    与 V1 的日志格式保持一致，便于横向对比
        log_retrieval_trace(
            logger,
            stage="v2_dense_retrieve",
            question=question,
            hits=dense_hits,
            extra={"candidate_k": candidate_k, "collection": settings.milvus_collection},
        )
        log_retrieval_trace(
            logger,
            stage="v2_sparse_retrieve",
            question=question,
            hits=sparse_hits,
            extra={"candidate_k": candidate_k, "collection": settings.milvus_collection},
        )

        # ④ RRF 融合（k 取自 settings.rrf_k，不硬编码）
        dense_ids = [h["chunk_id"] for h in dense_hits]
        sparse_ids = [h["chunk_id"] for h in sparse_hits]
        fused = rrf.fuse([dense_ids, sparse_ids], top_k=final_k)

        # ⑤ 记录融合结果及每个块在两路中的原始名次，供归因分析
        #    归因需要的「该块靠哪一路召回」信息就藏在 dense_rank / sparse_rank 中，
        #    0 表示该块未出现在对应通道
        dense_rank = {cid: i for i, cid in enumerate(dense_ids, start=1)}
        sparse_rank = {cid: i for i, cid in enumerate(sparse_ids, start=1)}
        log_retrieval_trace(
            logger,
            stage="v2_rrf_fuse",
            question=question,
            hits=[
                {
                    "chunk_id": cid,
                    "score": score,
                    "page_no": next(
                        (h["page_no"] for h in dense_hits + sparse_hits if h["chunk_id"] == cid),
                        "?",
                    ),
                }
                for cid, score in fused
            ],
            extra={
                "rrf_k": settings.rrf_k,
                "final_k": final_k,
                "dense_rank": dense_rank,
                "sparse_rank": sparse_rank,
            },
        )

        if not fused:
            return []

        # ⑥ 回填 MySQL 元数据（页码溯源真相源，ADR-005）
        fused_ids = [cid for cid, _ in fused]
        score_map = dict(fused)
        rows = mysql.fetch_chunks_by_ids(fused_ids)
        row_map = {r["chunk_id"]: r for r in rows}

        results: List[Dict[str, Any]] = []
        for cid in fused_ids:
            row = row_map.get(cid)
            if row is None:
                logger.warning("MySQL 中缺少 chunk 元数据，已跳过 | %s", cid)
                continue
            results.append(
                {
                    "chunk_id": cid,
                    "score": score_map.get(cid, 0.0),
                    "page_no": row.get("page_no", 1),
                    "page_nums": row.get("page_nums", []),
                    "file_name": row.get("source_file", ""),
                    "content": row.get("content", ""),
                    "content_type": row.get("content_type", "text"),
                    "section_title": row.get("section_title", ""),
                }
            )
        return results


_pipeline: Optional[V2Pipeline] = None


def get_pipeline() -> V2Pipeline:
    """获取 V2 链路单例（供 get_pipeline_by_name('v2') 调用）"""
    global _pipeline
    if _pipeline is None:
        _pipeline = V2Pipeline()
    return _pipeline
```
