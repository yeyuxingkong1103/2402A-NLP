# -*- coding: utf-8 -*-
"""批次 31 验收：真实起后端 → 走 SSE 跑一轮问答 → 查 chat_messages.model。

为什么这样设计（而不是复用 e2e_auth_persist.py）：
- 本批只改 `persist_turn` 的 model 取值，验收点单一：**落库的 model 是不是当轮配置值**；
- e2e_auth_persist.py 走"注册 → 邮件验证码 → 登录"全链路，依赖 SMTP 真发信，与本批无关；
- 这里直接用真 Redis 的 SessionStore 签发探针令牌（`chat_sessions.user_id` 只存 user_key、
  无外键到 users），因此**不必**创建 users 行，探针数据完全可控、可幂等清理。

硬性纪律：
- 端口 8124（8123 已被占用、8000 被 Docker Desktop 占用）；
- 子进程 stdout/stderr 重定向到日志文件（接 PIPE 无人消费会写满 64KB 缓冲卡死服务）；
- 先剔除 http_proxy/https_proxy（沙箱代理会劫持 127.0.0.1）；
- 收尾删干净探针数据（只按 PROBE_USER 删，不触碰其它行）。
"""

from __future__ import annotations

import json
import os
import random
import subprocess
import sys
import time
from pathlib import Path

PROJECT_ROOT = Path(r"C:/Users/92842/Desktop/rag")
BACKEND = PROJECT_ROOT / "backend"
PY = r"C:/Users/92842/anaconda3/envs/rag/python.exe"
TMP = Path(r"C:/Users/92842/AppData/Local/Temp/b31_bak")
LOG = TMP / "b31_server.log"
PORT = 8124
BASE = f"http://127.0.0.1:{PORT}"
# 探针用户：固定值便于幂等清理；session_key 每次随机，避免与历史会话纠缠
PROBE_USER = "e2e_b31_model_probe"
SESSION_KEY = f"e2e_b31_session_{random.randrange(10**8):08d}"

sys.path.insert(0, str(PROJECT_ROOT))
from scripts._env import load_project_env, mysql_config  # noqa: E402

load_project_env(PROJECT_ROOT)  # 含剔除代理变量
sys.path.insert(0, str(BACKEND))

import httpx  # noqa: E402
import pymysql  # noqa: E402

from app.auth.session_store import SessionStore  # noqa: E402
from app.core.config import settings  # noqa: E402
from app.db.redis_client import create_redis_client  # noqa: E402

RESULTS: list[tuple[str, bool, str]] = []


def check(name: str, ok: bool, detail: str = "") -> None:
    """登记一项验收结果并即时打印。"""
    RESULTS.append((name, ok, detail))
    print(f"{'✅' if ok else '❌'} {name}" + (f"\n     {detail}" if detail else ""))


def connect() -> pymysql.connections.Connection:
    """连真 MySQL（参数全部来自 .env，脚本内零明文）。"""
    return pymysql.connect(charset="utf8mb4", autocommit=True, **mysql_config())


def cleanup_probe(conn) -> None:
    """按 PROBE_USER 删除探针会话与消息（幂等；不动任何其它 user 的行）。"""
    with conn.cursor() as cur:
        cur.execute(
            "DELETE m FROM chat_messages m JOIN chat_sessions s ON m.session_id = s.id "
            "WHERE s.user_id = %s",
            (PROBE_USER,),
        )
        deleted_messages = cur.rowcount
        cur.execute("DELETE FROM chat_sessions WHERE user_id = %s", (PROBE_USER,))
        deleted_sessions = cur.rowcount
    if deleted_messages or deleted_sessions:
        print(f"[清理] 探针历史残留：消息 {deleted_messages} 行、会话 {deleted_sessions} 行")


def clear_probe_redis() -> None:
    """清掉探针用户的 Redis 令牌集合与本次令牌键。"""
    redis_client = create_redis_client(settings.redis_url)
    tokens = redis_client.smembers(f"user_sessions:{PROBE_USER}")
    for token in tokens:
        redis_client.delete(f"session:{token}")
    redis_client.delete(f"user_sessions:{PROBE_USER}")
    if tokens:
        print(f"[清理] 探针 Redis 令牌 {len(tokens)} 个")


def start_server() -> tuple[subprocess.Popen, object]:
    """起真后端；stdout/stderr 落日志文件（禁止 PIPE），返回进程与日志句柄。"""
    env = dict(os.environ)
    for key in ("http_proxy", "https_proxy", "HTTP_PROXY", "HTTPS_PROXY"):
        env.pop(key, None)
    log_handle = open(LOG, "wb")
    proc = subprocess.Popen(
        [PY, "-m", "uvicorn", "app.main:app", "--host", "127.0.0.1", "--port", str(PORT)],
        cwd=str(BACKEND),
        env=env,
        stdout=log_handle,
        stderr=subprocess.STDOUT,
    )
    return proc, log_handle


def wait_healthy(timeout: float = 90.0) -> bool:
    """轮询 /health/live 直到就绪（本端口只有我们起的进程能应答）。"""
    deadline = time.time() + timeout
    while time.time() < deadline:
        try:
            resp = httpx.get(f"{BASE}/health/live", timeout=2.0)
            if resp.status_code == 200:
                print(f"[服务] 就绪：{BASE}  server={resp.headers.get('server')}")
                return True
        except Exception:  # noqa: BLE001
            time.sleep(1.0)
    return False


def stream_chat(token: str) -> tuple[int, list[tuple[str, dict]], str, dict]:
    """打真实 SSE 接口，返回 (状态码, 事件列表, 拼接后的回答, 响应头)。"""
    payload = {
        "session_id": SESSION_KEY,
        "character_id": "legal-assistant",
        "message": "经济补偿怎么算",
        "options": {"top_k": 5},
    }
    events: list[tuple[str, dict]] = []
    answer_parts: list[str] = []
    headers: dict = {}
    status = 0
    with httpx.Client(timeout=httpx.Timeout(240.0, connect=10.0)) as client:
        with client.stream(
            "POST",
            f"{BASE}/api/v1/chat/stream",
            json=payload,
            headers={"Authorization": f"Bearer {token}"},
        ) as resp:
            status = resp.status_code
            headers = dict(resp.headers)
            event_name = None
            for line in resp.iter_lines():
                if line.startswith("event: "):
                    event_name = line[7:].strip()
                elif line.startswith("data: ") and event_name:
                    data = json.loads(line[6:])
                    events.append((event_name, data))
                    if event_name == "token":
                        answer_parts.append(data.get("text", ""))
                    if event_name == "replace":
                        answer_parts = [data.get("text", "")]
                    event_name = None
    return status, events, "".join(answer_parts), headers


def query_rows(conn, limit: int = 4) -> tuple[int, list[tuple]]:
    """取探针会话最新若干条消息（含 model 列原值）。"""
    with conn.cursor() as cur:
        cur.execute("SELECT id FROM chat_sessions WHERE session_key = %s", (SESSION_KEY,))
        row = cur.fetchone()
        if row is None:
            return 0, []
        session_id = row[0]
        cur.execute(
            "SELECT id, role, model, CHAR_LENGTH(content), created_at "
            "FROM chat_messages WHERE session_id = %s ORDER BY id DESC LIMIT %s",
            (session_id, limit),
        )
        return session_id, list(cur.fetchall())


def count_null_models(conn) -> int:
    """**排除探针会话**后 model 为 NULL 的行数（证明历史行未被回填）。

    为什么要排除探针会话：本轮新写入的 user 消息本来就该是 NULL
    （提问不由模型生成），若统计全表，NULL 数必然 +1，那是预期行为不是回填。
    """
    with conn.cursor() as cur:
        cur.execute(
            "SELECT COUNT(*) FROM chat_messages WHERE model IS NULL AND session_id NOT IN "
            "(SELECT id FROM chat_sessions WHERE user_id = %s)",
            (PROBE_USER,),
        )
        return int(cur.fetchone()[0])


def count_non_null_models(conn) -> int:
    """全表 model 非 NULL 的行数（本批后应只多出探针那一行）。"""
    with conn.cursor() as cur:
        cur.execute("SELECT COUNT(*) FROM chat_messages WHERE model IS NOT NULL")
        return int(cur.fetchone()[0])


def model_column_definition(conn) -> str:
    """取 model 列的 DDL 定义（确认落库值不会被列宽截断）。"""
    with conn.cursor() as cur:
        cur.execute("SHOW COLUMNS FROM chat_messages LIKE 'model'")
        row = cur.fetchone()
    return f"{row[0]} {row[1]}" if row else "(未找到 model 列)"


def main() -> int:
    conn = connect()
    print(f"[配置] MYSQL={mysql_config()['host']}:{mysql_config()['port']}/"
          f"{mysql_config()['database']}  LLM_MODEL={settings.llm_model}")
    print(f"[探针] user_id={PROBE_USER}  session_key={SESSION_KEY}")

    # 0) 幂等清理 + 基线
    cleanup_probe(conn)
    clear_probe_redis()
    null_before = count_null_models(conn)
    non_null_before = count_non_null_models(conn)
    print(f"[基线] 跑前：历史 model IS NULL 行数（排除探针会话）= {null_before}；"
          f"全表 model 非 NULL 行数 = {non_null_before}")
    print(f"[基线] model 列定义：{model_column_definition(conn)}")

    # 1) 签发探针令牌（真 Redis，与生产 login 同一 SessionStore）
    token = SessionStore(
        create_redis_client(settings.redis_url), settings.session_ttl_seconds
    ).create_session_token(PROBE_USER, False)
    print(f"[令牌] 已签发（长度 {len(token)}）")

    # 2) 起服务并等就绪
    proc, log_handle = start_server()
    try:
        if not wait_healthy():
            check("服务启动", False, f"未就绪，见 {LOG}")
            return 1
        check("服务启动", True, f"uvicorn 监听 {BASE}（日志 {LOG.name}）")

        # 3) 真 SSE 一轮问答
        status, events, answer, headers = stream_chat(token)
        names = [name for name, _ in events]
        check(
            "SSE 返回 200 且事件序列完整",
            status == 200 and "message_start" in names and "message_end" in names,
            f"status={status} 事件={ {n: names.count(n) for n in dict.fromkeys(names)} }",
        )
        check(
            "回答非空且有引用编号",
            bool(answer.strip()) and "[" in answer,
            f"回答长度={len(answer)}  首句={answer.strip().splitlines()[0][:60] if answer.strip() else '(空)'}",
        )

        # 4) 查库：最新行 model 有值
        session_id, rows = query_rows(conn)
        print("\n[落库] SELECT id, role, model, CHAR_LENGTH(content), created_at "
              "FROM chat_messages WHERE session_id=%s ORDER BY id DESC;" % session_id)
        for row in rows:
            print(f"   id={row[0]}  role={row[1]}  model={row[2]!r}  "
                  f"len={row[3]}  created_at={row[4]}")

        assistant = next((r for r in rows if r[1] == "assistant"), None)
        user_row = next((r for r in rows if r[1] == "user"), None)
        check(
            "assistant 消息 model 等于当轮配置模型名",
            assistant is not None and assistant[2] == settings.llm_model,
            f"落库 model={assistant[2]!r}  当轮配置={settings.llm_model!r}",
        )
        check(
            "assistant 消息 model 非 NULL",
            assistant is not None and assistant[2] is not None,
            f"model={assistant[2]!r}",
        )
        check(
            "user 消息仍为 NULL（只改 assistant 侧取值）",
            user_row is not None and user_row[2] is None,
            f"model={user_row[2] if user_row else '(缺行)'!r}",
        )
        check("本轮恰好落库 2 条消息", len(rows) == 2, f"实际 {len(rows)} 条")

        # 5) 历史行未被回填 / 未被猜测
        null_after = count_null_models(conn)
        non_null_after = count_non_null_models(conn)
        check(
            "历史 model IS NULL 行未被回填（排除探针会话统计）",
            null_after == null_before,
            f"跑前 {null_before} → 跑后 {null_after}",
        )
        check(
            "全表非 NULL 行只多出本轮的 assistant 那一行",
            non_null_after - non_null_before == 1,
            f"跑前 {non_null_before} → 跑后 {non_null_after}",
        )
    finally:
        proc.terminate()
        try:
            proc.wait(timeout=20)
        except subprocess.TimeoutExpired:
            proc.kill()
        log_handle.close()
        # 6) 收尾清理探针数据
        cleanup_probe(conn)
        clear_probe_redis()
        conn.close()

    # 7) 收尾核对：探针数据（含历史残留）已清零
    residue_conn = connect()
    with residue_conn.cursor() as cur:
        cur.execute("SELECT COUNT(*) FROM chat_sessions WHERE user_id = %s", (PROBE_USER,))
        residue_sessions = int(cur.fetchone()[0])
        cur.execute(
            "SELECT COUNT(*) FROM chat_messages m JOIN chat_sessions s ON m.session_id = s.id "
            "WHERE s.user_id = %s",
            (PROBE_USER,),
        )
        residue_messages = int(cur.fetchone()[0])
    residue_conn.close()
    check(
        "探针数据已清干净（会话/消息残留均为 0）",
        residue_sessions == 0 and residue_messages == 0,
        f"残留会话 {residue_sessions} 行、消息 {residue_messages} 行",
    )

    passed = sum(1 for _, ok, _ in RESULTS if ok)
    print(f"\n=== 汇总：{passed}/{len(RESULTS)} 通过 ===")
    for name, ok, _ in RESULTS:
        print(f"  {'✅' if ok else '❌'} {name}")
    return 0 if passed == len(RESULTS) else 1


if __name__ == "__main__":
    sys.exit(main())
