```python
    uid = redis.get(f"session:token:{x_token}")
    if uid is None: raise HTTPException(401, "登录已过期，请重新登录")
```

```python
    def set_password(self, password):
        salt = secrets.token_hex(16)
        self.password_hash = f"{salt}${self._hash(password, salt)}"
    def verify_password(self, password):
        salt, digest = self.password_hash.split("$", 1)
        return secrets.compare_digest(self._hash(password, salt), digest)

        return hashlib.sha256(f"{salt}{password}".encode("utf-8")).hexdigest()
```

```python
def chunk_text(text, chunk_size=700, overlap=80):
    sentences = split_sentences(text)
    for sentence in sentences:
        if len(sentence) > chunk_size:      # 单句超长 → 窗口硬切兜底
```

```python
def _cosine(a, b):
    dot = sum(x*y for x, y in zip(a, b))
    na = sum(x*x for x in a) ** 0.5
    nb = sum(y*y for y in b) ** 0.5
    return dot / (na * nb) if na and nb else 0.0
```
解析：余弦相似度——向量点积除以模长之积，用于语义分块与长期记忆过滤两处。

```python
class BM25Encoder:
    def fit(self, texts):
        tokenized = [self.tokenizer(t) for t in texts]
        doc_freq = {}                      # 词 → 出现文档数
        for tokens in tokenized:
            for term in set(tokens): doc_freq[term] = doc_freq.get(term, 0) + 1
        self.vocab = {term: i for i, term in enumerate(sorted(doc_freq))}
        self.idf = {term: math.log((n - df + 0.5)/(df + 0.5)) + 1.0 ...}
        self.avg_len = sum(self.doc_len)/n
```
解析：语料统计——jieba 分词后统计文档频率 DF、构建词表（词→索引）、算 IDF（罕见词权重高）、记录平均文档长度。

```python
class LazyModelProxy:
    def __init__(self, factory): self._factory = factory; self._instance = None; self._lock = threading.Lock()
    def _get(self):
        if self._instance is None:
            with self._lock:               # 双重检查锁：并发下只加载一次
```
