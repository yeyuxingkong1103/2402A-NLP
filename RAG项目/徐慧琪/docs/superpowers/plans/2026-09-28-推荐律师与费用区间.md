```python
"""案由标签与专业领域映射。

为什么用纯规则而不是模型（技术方案 6.4 定稿）：案由是下游「推荐律师」与
「费用区间」的共同输入，判错会把用户导向错误的领域与收费口径。规则可穷举
测试、可解释、零成本；模型判错时没有便宜的发现手段。
"""
from __future__ import annotations

import re

# 案由认不出时的兜底领域：不给具体领域，但律师卡片照常展示（设计 §七）
GENERIC_FIELD = "通用"

# 案由规则。**顺序是语义的一部分**：先具体后笼统，命中即返回。
# 例如「离婚后房子归谁」同时含「离婚」与「房子」，必须判成婚姻家庭而非租赁。
CAUSE_RULES: list[tuple[re.Pattern[str], str]] = [
    (re.compile(r"(离婚|抚养|抚养费|夫妻共同财产|婚前财产)"), "婚姻家庭纠纷"),
    (re.compile(r"(拖欠工资|加班费|经济补偿|违法辞退|劳动合同|社保|工伤)"), "劳动争议"),
    (re.compile(r"(网购|假货|退货|退款|三倍赔偿|消费者)"), "消费者权益纠纷"),
    (re.compile(r"(借款|借钱|欠条|借条|利息|民间借贷)"), "民间借贷纠纷"),
    (re.compile(r"(租房|押金|房东|承租人|租金|退租)"), "房屋租赁合同纠纷"),
]

# 案由 → 专业领域。演示用映射表，正式运营前随真实律师名录一起替换
CAUSE_TO_FIELD: dict[str, str] = {
    "婚姻家庭纠纷": "婚姻家庭",
    "劳动争议": "劳动法",
    "消费者权益纠纷": "消费者权益",
    "民间借贷纠纷": "金融与借贷",
    "房屋租赁合同纠纷": "房产与建设工程",
}


def tag(question: str) -> str | None:
    """问句 → 案由标签；认不出返回 None（不猜）。"""
    text = question or ""
    for pattern, cause in CAUSE_RULES:
        if pattern.search(text):
            return cause
    return None


def field_of(cause: str | None) -> str:
    """案由 → 专业领域；认不出给 GENERIC_FIELD（卡片仍能展示，见设计 §七）。"""
    if cause is None:
        return GENERIC_FIELD
    return CAUSE_TO_FIELD.get(cause, GENERIC_FIELD)
```

"""示例律师卡片生成。

为什么随机生成（技术方案 6.4 定稿）：不接真实名录，仅用于演示与联调。
**对外正式运营前必须替换为真实在册律师信息**，否则构成虚假宣传（风险 11）——
所以 DEMO_MODE 为真时卡片必须带「示例数据」标注，这条有测试钉住。
"""
from __future__ import annotations

import random

# 演示开关：接入真实名录时改这里，并把 DEMO_NOTE 从卡片上撤下
DEMO_MODE = True
DEMO_NOTE = "示例数据"

# 演示期固定的联系方式获取方式：不产出任何真实联系方式
CONTACT_HINT = "通过平台留言获取"

# 姓名与机构名用字表。只用常见字，避免生成生僻到像乱码的名字
SURNAMES = "赵钱孙李周吴郑王冯陈褚卫蒋沈韩杨朱秦许何吕施张孔曹严华金魏陶姜"
GIVEN_CHARS = "文伟芳娜敏静秀丽强磊洋勇艳杰娟涛明超霞平刚桂英建国民"
ORG_PREFIX = "明理正衡德信公诚瑞达弘正同仁经纶格物"
ORG_SUFFIX = "律师事务所"


def _name(rng: random.Random) -> str:
    """常见姓 + 1~2 个名用字。"""
    given = "".join(rng.choice(GIVEN_CHARS) for _ in range(rng.randint(1, 2)))
    return rng.choice(SURNAMES) + given


def _org(rng: random.Random) -> str:
    return rng.choice(ORG_PREFIX) + rng.choice(ORG_PREFIX) + ORG_SUFFIX


def recommend(field: str, n: int = 3, rng: random.Random | None = None) -> list[dict]:
    """生成 n 张示例律师卡片。

    排序只由领域匹配度决定（此处即「卡片领域 == 请求领域」），
    没有任何外部输入能改变顺序 —— FR-9.5 / AC-21 的「无付费插位」。
    rng 作为参数而非模块全局：测试要能复现同一批卡片。
    """
    rng = rng or random.Random()
    cards = []
    for _ in range(n):
        card = {
            "name": _name(rng),
            "org": _org(rng),
            "field": field,
            "contact_hint": CONTACT_HINT,
            "demo": DEMO_MODE,
        }
        if DEMO_MODE:
            # 标注与卡片同层，渲染方不必再判断一次开关
            card["note"] = DEMO_NOTE
        cards.append(card)
    return cards
```

"""费用区间：数值回查 + 生成编排（编排部分见本文件后半）。

技术方案 6.4 的两条硬规矩在这里落地：
  ①区间**不得脱离命中片段编造** —— 由 verify_range 保证「数字有字面出处」；
  ②命中不到**不猜数** —— 由编排层的 no_corpus 分支保证。
回查是**必要不充分**条件：它挡的是「模型报了个片段里没有的数」，
挡不住「模型把片段里的数用错了地方」——后者靠提示词约束与人工冒烟。

from __future__ import annotations

import re

# 只认阿拉伯数字。中文数字（「百分之八」）不抽 —— 语料与模型输出都约定用
# 阿拉伯数字写法，宁可拒掉也不做易错的等价换算
_NUMBER = re.compile(r"\d+(?:\.\d+)?")


def numbers_in(text: str) -> set[str]:
    """抽出文本里所有阿拉伯数字的字面写法（不做单位与量纲换算）。"""
    if not text:
        return set()
    return {m.group(0) for m in _NUMBER.finditer(text)}


def _literal(value: float) -> str:
    """把数值写成与片段比对用的字面量：8.0 与 8 视为同一个数。"""
    number = float(value)
    return str(int(number)) if number.is_integer() else str(number)


def verify_range(low: float | None, high: float | None, snippet: str) -> bool:
    """区间的两个端点是否都能在片段里找到字面出处。

    缺失端点、空片段一律判否 —— 宁可输出「暂无费用口径依据」，
    也不输出一个无法指回片段的数字（技术方案 6.4 硬规矩①）。
    """
    if low is None or high is None:
        return False
    pool = numbers_in(snippet)
    if not pool:
        return False
    return _literal(low) in pool and _literal(high) in pool

# AC-21 定稿原话，渲染方直接取用，不得改写
FEE_DISCLAIMER = "参考区间，不构成报价或委托"


def estimate(cause: str, *, search_fn, generate_fn, log_fn) -> dict:
    """在 fee_corpus 上取一段收费口径，据它产出区间。

    三个依赖全部注入：search_fn(cause) -> list[dict]、generate_fn(snippet, cause)
    -> {"low", "high"}、log_fn(**fields) 留痕（AC-21 证据）。

    **故障一律向上传播**：检索或模型不可用时不返回任何状态，由调用方
    区分「服务不可用」与「语料里确实没有」——两者给用户的含义完全不同。
    """
    hits = search_fn(cause)
    if not hits:
        return {"status": "no_corpus", "low": None, "high": None,
                "basis": None, "source_doc": None, "source_no": None,
                "reason": "语料未收录该案由的收费口径"}
    snippet = hits[0]
    produced = generate_fn(snippet["text"], cause)
    low, high = produced.get("low"), produced.get("high")
    common = {"cause": cause, "snippet_id": snippet.get("source_no"),
              "model": produced.get("model"), "request_id": produced.get("request_id")}
    if not verify_range(low, high, snippet["text"]):
        # 被拒的区间也要留痕：AC-21 要的是可追溯，包括「拒绝过什么」
        log_fn(**common, range_low=low, range_high=high, status="rejected")
        return {"status": "rejected", "low": None, "high": None,
                "basis": None, "source_doc": None, "source_no": None,
                "reason": "生成区间无法指回命中片段，已按不猜数处理"}
    log_fn(**common, range_low=low, range_high=high, status="ok")
    return {"status": "ok", "low": low, "high": high,
            "basis": snippet["text"], "source_doc": snippet.get("source_doc"),
            "source_no": snippet.get("source_no"), "reason": None}
```

AC-19 约束：本表不记录用户问题原文、IP 或任何身份信息，故表结构里

from __future__ import annotations

# ts 用数据库默认值：避免应用时钟与数据库时钟不一致
DDL = """
CREATE TABLE IF NOT EXISTS fee_generation_log (
  id BIGINT AUTO_INCREMENT PRIMARY KEY,
  ts DATETIME NOT NULL DEFAULT CURRENT_TIMESTAMP,
  cause VARCHAR(64) NULL COMMENT '案由标签；刻意不存用户问题原文（AC-19）',
  snippet_id VARCHAR(128) NULL COMMENT '命中的收费口径片段 id',
  range_low DECIMAL(12,2) NULL,
  range_high DECIMAL(12,2) NULL,
  model VARCHAR(64) NULL,
  request_id VARCHAR(64) NULL,
  status VARCHAR(16) NOT NULL COMMENT 'ok / rejected',
  INDEX idx_ts (ts)
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4
"""

# 插入列的固定顺序：record 与 make_logger 共用，改这里就够
_COLUMNS = ("cause", "snippet_id", "range_low", "range_high",
            "model", "request_id", "status")


def ensure_table(conn) -> None:
    """建表（幂等）—— 启动时无条件调用。"""
    with conn.cursor() as cur:
        cur.execute(DDL)
    conn.commit()


def record(conn, **fields) -> None:
    """写一行留痕。只认 _COLUMNS 里的字段，缺的记 NULL。"""
    placeholders = ", ".join(["%s"] * len(_COLUMNS))
    values = tuple(fields.get(name) for name in _COLUMNS)
    with conn.cursor() as cur:
        cur.execute(
            f"INSERT INTO fee_generation_log ({', '.join(_COLUMNS)}) "
            f"VALUES ({placeholders})", values)
    conn.commit()


def make_logger(conn):
    """返回可直接传给 fees.estimate(log_fn=...) 的可调用对象。"""
    def log(**fields) -> None:
        record(conn, **fields)
    return log

"""附加区块组装。

存在的理由：把「案由 → 律师 → 费用」三块收成一个纯组装点，供问答编排层
一次调用。它是**故障降级**的落点：费用那块挂掉（检索/模型不可用）时标成
unavailable 并继续返回，绝不把服务故障说成「暂无费用口径依据」——
那是 ③a 定下的红线。
"""
from __future__ import annotations

from app.recommend import cause as cause_mod
from app.recommend import fees, lawyers

# 默认给三张卡片：FR-9.3 要求「按领域匹配展示」，1~3 张是技术方案 6.4 定的范围
DEFAULT_CARD_COUNT = 3


def _unavailable(exc: Exception) -> dict:
    """故障态的 fee 结果 —— 与 no_corpus 严格区分（渲染文案不同）。"""
    return {"status": "unavailable", "low": None, "high": None, "basis": None,
            "source_doc": None, "source_no": None,
            "reason": f"费用信息暂时不可用：{exc}"}


def build(question: str, *, search_fn, generate_fn, log_fn, rng=None) -> dict:
    """组装附加区块。三个依赖全部注入（见 Task 4/5 的接口）。

    留痕（log_fn）失败**吞掉**：审计是旁路，不能让用户因此看不到费用信息；
    检索与生成的失败则转成 unavailable —— 两种失败的归属不同。
    """
    cause_name = cause_mod.tag(question)
    field = cause_mod.field_of(cause_name)
    cards = lawyers.recommend(field, DEFAULT_CARD_COUNT, rng)

    logged = []

    def guarded_log(**fields):
        # 记下最后一次留痕尝试的失败，但不打断费用流程
        try:
            log_fn(**fields)
        except Exception as exc:  # noqa: BLE001 —— 留痕失败不改变用户体验
            logged.append(exc)

    try:
        fee = fees.estimate(cause_name or "", search_fn=search_fn,
                            generate_fn=generate_fn, log_fn=guarded_log)
    except Exception as exc:  # noqa: BLE001 —— 检索/生成故障：降级但不伪装
        fee = _unavailable(exc)

    return {"cause": cause_name, "field": field, "lawyers": cards,
            "fee": fee, "disclaimer": fees.FEE_DISCLAIMER}

def format_extras(block: dict | None) -> str:
    """渲染附加区块。没给区块（律师侧 / 非 ok）就返回空串，调用方直接拼接。"""
    if not block:
        return ""
    lines = [f"\n【专业领域】{block['field']}"]
    if block.get("cause"):
        lines.append(f"【案由】{block['cause']}")
    if block["lawyers"]:
        lines.append("【示例律师】（演示数据）")
        for card in block["lawyers"]:
            note = f"（{card['note']}）" if card.get("note") else ""
            lines.append(f"  - {card['name']}{note}｜{card['org']}｜"
                         f"{card['field']}｜{card['contact_hint']}")
    fee = block.get("fee") or {}
    if fee.get("status") == "ok":
        lines.append(f"【费用区间】{fee['low']} ~ {fee['high']}"
                     f"（依据：{fee.get('source_doc')} {fee.get('source_no')}）")
    elif fee.get("status") == "no_corpus":
        lines.append("【费用区间】暂无费用口径依据")
    elif fee.get("status") == "rejected":
        lines.append("【费用区间】生成结果无法指回收费口径，已按不猜数处理")
    elif fee.get("status") == "unavailable":
        lines.append(f"【费用区间】{fee.get('reason')}")
    lines.append(f"【提示】{block['disclaimer']}")
    return "\n".join(lines)

from __future__ import annotations

import pathlib
import re
import sys

sys.path.insert(0, str(pathlib.Path(__file__).resolve().parents[1] / "backend"))

FEE_COLLECTION = "fee_corpus"
# 与法条一致的切分口径：条款号为界；无条款号的文档退化为按段
_ARTICLE = re.compile(r"^第[一二三四五六七八九十百零]+条", re.MULTILINE)


def split_paragraphs(text: str) -> list[str]:
    """按空行切段并去掉空白块。"""
    return [chunk.strip() for chunk in re.split(r"\n\s*\n", text or "")
            if chunk.strip()]


def build_rows(doc_path: pathlib.Path, region: str, effective_date: str) -> list[dict]:
    """把一份文档变成待入库的行。source_no 取段落开头的条号（有则取，无则空）。"""
    text = doc_path.read_text(encoding="utf-8")
    rows = []
    for unit in split_paragraphs(text):
        match = _ARTICLE.match(unit)
        rows.append({
            "text": unit,
            "source_doc": doc_path.stem,
            "source_no": match.group(0) if match else "",
            "region": region,
            "effective_date": effective_date,
            "status": "现行有效",
        })
    return rows

    # 具体编码与写库沿用 app/ingest/embed.py 与 app/db/milvus.py 的既有写法
