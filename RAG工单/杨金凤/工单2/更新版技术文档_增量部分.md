## 3. 关键实现（新增）

### 3.1 Reranker 精排

```python
def _rerank(self, query, docs, top_k=5):
    pairs = [[query, d.page_content] for d in docs]
    with torch.no_grad():
        inputs = self.reranker_tokenizer(
            pairs, padding=True, truncation=True,
            return_tensors="pt", max_length=512,
        ).to(self.reranker_model.device)
        scores = self.reranker_model(**inputs).logits.view(-1).float()
    ranked = sorted(zip(scores.tolist(), docs), key=lambda x: x[0], reverse=True)
    return [d for _, d in ranked[:top_k]]
```

### 3.2 语言检测

```python
def _detect_language(self, text):
    zh_chars = len(re.findall(r"[\u4e00-\u9fff]", text))
    total = len(re.sub(r"\s", "", text))
    return "zh" if zh_chars / total >= 0.3 else "en"
```

### 3.3 输入校验

```python
def _validate_input(self, question):
    if question is None:
        return False, "", "输入为空，请输入问题。"
    cleaned = re.sub(r"\s+", " ", question).strip()
    if len(cleaned) < 1:
        return False, "", "输入为空，请输入有效问题。"
    if len(cleaned) > 500:
        cleaned = cleaned[:500]
        return True, cleaned, "问题过长，已自动截断。"
    if not re.search(r"[\u4e00-\u9fffA-Za-z0-9]", cleaned):
        return False, "", "输入不包含有效字符。"
    return True, cleaned, ""
```
